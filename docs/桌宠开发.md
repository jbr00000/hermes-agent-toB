# AI Agent 桌宠模块 — 设计文档

> **实施说明（2026-08-23）**：本文档 §2 的 PySide6 桌面窗口路线与当前产品架构
> （无头 FastAPI 服务 + 浏览器前端，无同进程 Python GUI）不匹配，未采用。
> 实际落地为 **Web 内桌宠**（`apps/web/src/components/pet/`）：
>
> - 形象：铅笔小猫（程序化 SVG，`PetAvatar`）+ 牛来（原图去水印/抠图立绘
>   `assets/niulai.png` + CSS 程序化动画，`NiulaiAvatar`），`PetCharacter`
>   按 `petSkinAtom` 统一分发；素材由 `scripts/build_pet_asset.py` 一次性生成。
> - 状态机：沿用前端既有 `PetState`（idle/thinking/working/confused/eureka/sad），
>   由 chat/agent 运行状态推导（`petState.ts`），语义与本文 §4 的任务态对应。
> - 交互：拖拽移动（位置持久化 localStorage）、点击挤压弹跳反应。
> - 管理面板：顶栏 `PetSettingsButton` 弹出层——显示开关、形象选择、大小滑杆、
>   待机微动效 / 任务动画开关、六状态预览（含触发时机说明），全部
>   `atomWithStorage` 持久化。
>
> 以下原文档的状态机设计、动画素材规范、交互需求（F-01~F-06）作为设计参考保留。

## 1. 任务描述

### 1.1 项目背景

本模块为自研 AI Agent 智能体的**可视化表现层**，以"桌宠"形式运行在用户桌面上，
通过动画反馈 Agent 的实时工作状态，提升用户交互体验。

### 1.2 功能需求

| 编号 | 功能 | 描述 |
|------|------|------|
| F-01 | 透明置顶窗口 | 桌宠悬浮在桌面最上层，无背景、无边框、不遮挡正常操作 |
| F-02 | 鼠标拖拽 | 用户可按住桌宠拖动到任意位置 |
| F-03 | 点击动画 | 单击/双击桌宠时播放反应动画 |
| F-04 | 任务动画 | Agent 执行不同任务时，桌宠播放对应的专属动画 |
| F-05 | 结果反馈 | 任务成功/失败时播放不同的反馈动画 |
| F-06 | 自主行为 | 无任务时桌宠有随机待机行为（走动、打瞌睡等） |
| F-07 | 系统托盘 | 提供右键菜单：隐藏、退出等基础操作 |

### 1.3 非功能需求

- **轻量**：内存占用 < 100MB，CPU 空闲时 < 2%
- **无对话功能**：桌宠仅作为视觉表现层，不承担对话交互
- **与 Agent 同进程**：通过信号机制通信，无需网络开销
- **跨平台**：支持 Windows / macOS / Linux

## 2. 技术栈

| 层级 | 选型 | 版本要求 | 选型理由 |
|------|------|---------|---------|
| 语言 | Python | ≥ 3.10 | 与现有 Agent 后端/算法端统一技术栈 |

## 4. 状态机设计

### 4.1 状态定义

```python
class PetState(Enum):
    IDLE        = "idle"         # 待机
    WALK        = "walk"         # 随机走动
    DRAG        = "drag"         # 被拖拽中
    CLICK       = "click"        # 被点击
    TASK_SEARCH = "task_search"  # 执行搜索任务
    TASK_CODE   = "task_code"    # 执行代码生成任务
    TASK_ANALYZE= "task_analyze" # 执行数据分析任务
    TASK_DONE   = "task_done"    # 任务完成
    TASK_ERROR  = "task_error"   # 任务出错
    SLEEP       = "sleep"        # 长时间无操作
```

### 4.2 状态转换图

```
                     超时(60s无操作)
         ┌──────────────────────────────────┐
         │                                  ▼
    ┌────┴────┐   双击     ┌───────┐    ┌───────┐
    │  IDLE   │──────────→│ CLICK │    │ SLEEP │
    └────┬────┘           └───┬───┘    └───┬───┘
         │                    │ 播完自动     │ 点击唤醒
         │                    ▼             │
         │                回到 IDLE ◄───────┘
         │
         │  Agent 下发任务信号
         ▼
    ┌──────────┐   完成信号   ┌───────────┐
    │ TASK_X   │────────────→│ TASK_DONE │──→ 回到 IDLE
    └────┬─────┘             └───────────┘
         │ 失败信号
         ▼
    ┌────────────┐
    │ TASK_ERROR │──→ 回到 IDLE
    └────────────┘

    任意状态 + 鼠标按下 → DRAG → 鼠标释放 → IDLE
```

### 4.3 转换规则表

| 当前状态 | 触发条件 | 目标状态 | 备注 |
|---------|---------|---------|------|
| 任意 | 鼠标按下拖拽 | DRAG | 最高优先级 |
| DRAG | 鼠标释放 | IDLE | — |
| IDLE | 双击 | CLICK | 播放完自动回 IDLE |
| IDLE | Agent 任务信号 | TASK_X | 循环播放直到任务结束 |
| TASK_X | 任务完成信号 | TASK_DONE | 播放完自动回 IDLE |
| TASK_X | 任务失败信号 | TASK_ERROR | 播放完自动回 IDLE |
| IDLE | 60s 无操作 | SLEEP | — |
| SLEEP | 任意点击 | IDLE | 唤醒 |
| IDLE | 随机触发 | WALK | 自主行为 |

## 6. 核心代码实现

### 6.1 动画加载器

```python
# desktop_pet/animation.py
from pathlib import Path
from PySide6.QtGui import QPixmap


class AnimationLoader:
    """加载并缓存帧动画序列"""

    def __init__(self, assets_dir: str):
        self.assets_dir = Path(assets_dir)
        self.cache: dict[str, list[QPixmap]] = {}

    def load(self, state_name: str) -> list[QPixmap]:
        if state_name in self.cache:
            return self.cache[state_name]

        folder = self.assets_dir / state_name
        if not folder.exists():
            return []

        files = sorted(folder.glob("*.png"), key=lambda f: int(f.stem))
        frames = [QPixmap(str(f)) for f in files]

        self.cache[state_name] = frames
        return frames
```

### 6.2 桌宠主窗口

```python
# desktop_pet/pet_window.py
import random
from PySide6.QtWidgets import QWidget, QLabel, QMenu
from PySide6.QtCore import Qt, QTimer, QPoint
from .animation import AnimationLoader
from .state_machine import PetState


class PetWindow(QWidget):

    def __init__(self, assets_dir: str = "desktop_pet/assets"):
        super().__init__()

        # ---------- 窗口属性 ----------
        self.setWindowFlags(
            Qt.FramelessWindowHint
            | Qt.WindowStaysOnTopHint
            | Qt.Tool
        )
        self.setAttribute(Qt.WA_TranslucentBackground)
        self.setFixedSize(200, 200)

        # 初始位置：屏幕右下角
        screen = self.screen().availableGeometry()
        self.move(screen.width() - 250, screen.height() - 250)

        # ---------- 动画 ----------
        self.loader = AnimationLoader(assets_dir)
        self.label = QLabel(self)
        self.label.setAlignment(Qt.AlignCenter)

        self.current_state = PetState.IDLE
        self.current_frame = 0
        self.is_looping = True

        self.anim_timer = QTimer(self)
        self.anim_timer.timeout.connect(self._next_frame)
        self.anim_timer.start(150)

        self.revert_timer = QTimer(self)
        self.revert_timer.setSingleShot(True)
        self.revert_timer.timeout.connect(
            lambda: self.set_state(PetState.IDLE)
        )

        # ---------- 拖拽 ----------
        self.is_dragging = False
        self.drag_offset = QPoint()

        # ---------- 随机行为 ----------
        self.behavior_timer = QTimer(self)
        self.behavior_timer.timeout.connect(self._random_behavior)
        self.behavior_timer.start(5000)

        self.set_state(PetState.IDLE)

    # ==================== 状态管理 ====================

    def set_state(self, state: PetState, duration_ms: int = 0):
        self.current_state = state
        self.current_frame = 0
        self.is_looping = (duration_ms == 0)

        if duration_ms > 0:
            self.revert_timer.start(duration_ms)
        else:
            self.revert_timer.stop()

        self._render_frame()

    # ==================== 渲染 ====================

    def _next_frame(self):
        frames = self.loader.load(self.current_state.value)
        if not frames:
            return

        self.current_frame += 1
        if self.current_frame >= len(frames):
            self.current_frame = 0 if self.is_looping else len(frames) - 1

        self._render_frame()

    def _render_frame(self):
        frames = self.loader.load(self.current_state.value)
        if frames and self.current_frame < len(frames):
            self.label.setPixmap(frames[self.current_frame])

    # ==================== 鼠标交互 ====================

    def mousePressEvent(self, event):
        if event.button() == Qt.LeftButton:
            self.is_dragging = True
            self.drag_offset = event.globalPosition().toPoint() - self.pos()
            self.set_state(PetState.DRAG)

    def mouseMoveEvent(self, event):
        if self.is_dragging:
            self.move(event.globalPosition().toPoint() - self.drag_offset)

    def mouseReleaseEvent(self, event):
        if event.button() == Qt.LeftButton and self.is_dragging:
            self.is_dragging = False
            self.set_state(PetState.IDLE)

    def mouseDoubleClickEvent(self, event):
        if not self.is_dragging:
            self.set_state(PetState.CLICK, duration_ms=800)

    # ==================== 右键菜单 ====================

    def contextMenuEvent(self, event):
        menu = QMenu(self)
        menu.addAction("隐藏", self.hide)
        menu.addAction("退出", self.close)
        menu.exec(event.globalPos())

    # ==================== 随机行为 ====================

    def _random_behavior(self):
        if self.current_state not in (PetState.IDLE, PetState.WALK):
            return

        r = random.random()
        if r < 0.6:
            pass
        elif r < 0.85:
            self._do_walk()
        else:
            self.set_state(PetState.SLEEP)

    def _do_walk(self):
        self.set_state(PetState.WALK)
        direction = random.choice([-1, 1])
        steps = random.randint(3, 8)
        screen = self.screen().availableGeometry()

        for _ in range(steps):
            new_x = self.x() + direction * 10
            new_x = max(0, min(new_x, screen.width() - self.width()))
            self.move(new_x, self.y())

        self.revert_timer.start(steps * 150 + 200)

    # ==================== 外部接口 ====================

    def notify_task_start(self, task_name: str):
        state_map = {
            "search":  PetState.TASK_SEARCH,
            "code":    PetState.TASK_CODE,
            "analyze": PetState.TASK_ANALYZE,
        }
        state = state_map.get(task_name, PetState.IDLE)
        self.set_state(state)

    def notify_task_done(self):
        self.set_state(PetState.TASK_DONE, duration_ms=1200)

    def notify_task_error(self):
        self.set_state(PetState.TASK_ERROR, duration_ms=1500)
```

### 6.3 Agent 信号桥

```python
# bridge.py
from PySide6.QtCore import QObject, Signal


class AgentPetBridge(QObject):
    """Agent → 桌宠 的信号桥（线程安全）"""

    task_started  = Signal(str)   # 参数：任务类型名
    task_finished = Signal()
    task_failed   = Signal()

    def __init__(self, pet_window):
        super().__init__()
        self.pet = pet_window
        self.task_started.connect(self.pet.notify_task_start)
        self.task_finished.connect(self.pet.notify_task_done)
        self.task_failed.connect(self.pet.notify_task_error)
```

## 7. Agent 集成方式

### 7.1 在 Agent 任务代码中调用

```python
class MyAgent:
    def __init__(self, pet_bridge: AgentPetBridge):
        self.bridge = pet_bridge

    async def execute_search(self, query: str):
        self.bridge.task_started.emit("search")
        try:
            result = await self.do_search(query)
            self.bridge.task_finished.emit()
            return result
        except Exception:
            self.bridge.task_failed.emit()
            raise
```
